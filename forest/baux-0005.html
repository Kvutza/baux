import baux

bounds = np.array([[-5.0, 5.0], [-5.0, 5.0]])

1. PyTree & Non-Euclidean Spaces 

import math
import numpy as np
import baux

space = {
    "scale": baux.Log(1e-4, 1e-1),
    "phase": baux.Periodic(0.0, 2.0 * math.pi),
    "mixture": baux.Simplex(4),
    "kernel": baux.categorical(["matern", "rbf", "rational"]),
}
target = np.array([0.4, 0.3, 0.2, 0.1])
penalty = {"matern": 0.0, "rbf": 0.05, "rational": 0.1}


def objective(p):
    scale_err = (math.log10(p["scale"]) + 2.0) ** 2
    phase_err = 1.0 - math.cos(p["phase"] - math.pi / 3.0)
    mix_err = float(np.sum((np.sqrt(p["mixture"]) - np.sqrt(target)) ** 2))
    return -(scale_err + phase_err + mix_err + penalty[p["kernel"]])


state = baux.init(space, baux.turbo_enn(), key=baux.key(0))
state, batch = baux.ask(state, num_arms=4)
y = np.array([objective(p) for p in batch.params])
state, mu = baux.tell(state, batch, y)
print(state.best_params, state.best_y)

2. Scanned & Parallel Rollouts (scan / pmap) 

import numpy as np
import baux

bounds = np.array([[-5.0, 5.0], [-5.0, 5.0]])
s0 = baux.init(bounds, baux.turbo_enn(), key=baux.key(7))

final_state, next_key, traces = baux.scan(
    s0,
    baux.key(7),
    steps=20,
    eval_fn=lambda x: -np.sum(x**2, axis=1),
    num_arms=4,
)

branches = baux.pmap(
    s0,
    baux.key(99),
    branches=4,
    branch_fn=lambda i, st, k: baux.scan(
        st, k, steps=10, eval_fn=lambda x: -np.sum(x**2, axis=1), num_arms=2
    )[0].best_y,
)

3. Functional ENN Surrogate 

import numpy as np
import baux

rng = np.random.default_rng(0)
train_x = rng.uniform(-2.0, 2.0, size=(32, 3))
train_y = -np.sum(train_x**2, axis=1, keepdims=True)
query_x = rng.uniform(-2.0, 2.0, size=(8, 3))

k0, k1 = baux.split(baux.key(42), 2)
enn_state = baux.enn_init(train_x, train_y, scale_x=True)
enn_state = baux.with_geometry(enn_state, "hyperbolic", local_k=4)

params = baux.fit(enn_state, k=10, key=k0)
post = baux.posterior(enn_state, query_x, params)
draws, idx = baux.sample(enn_state, query_x, params, key=k1, num_draws=16)
